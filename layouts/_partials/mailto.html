{{- /* Returns a mailto: href. Context: dict subject, body (both optional). urlquery encodes spaces as "+", which mail clients don't decode in mailto: → use %20. */ -}} {{- $href := printf "mailto:%s" site.Params.email -}} {{- $q := slice -}} {{- with .subject }}{{ $q = $q | append (printf "subject=%s" (replace (urlquery .) "+" "%20")) }}{{ end -}} {{- with .body }}{{ $q = $q | append (printf "body=%s" (replace (urlquery .) "+" "%20")) }}{{ end -}} {{- with $q }}{{ $href = printf "%s?%s" $href (delimit . "&") }}{{ end -}} {{- return $href -}}
